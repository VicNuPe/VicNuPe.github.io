    function renderPalette() {
      paletteGrid.innerHTML = "";

      // Header column labels
      const headerRow = document.createElement("div");
      headerRow.className = "palette-row";
      const emptyLabel = document.createElement("div");
      emptyLabel.className = "palette-family-label";
      emptyLabel.textContent = "Shades";
      headerRow.appendChild(emptyLabel);

      const headerSwatches = document.createElement("div");
      headerSwatches.className = "palette-swatches";
      standardShades.forEach(shade => {
        const span = document.createElement("div");
        span.className = "w-8 text-center text-[10px] font-bold text-slate-400";
        span.textContent = shade;
        headerSwatches.appendChild(span);
      });
      // Extra for black/white section column alignment
      const bwHeaderSpan = document.createElement("div");
      bwHeaderSpan.className = "w-20 text-center text-[10px] font-bold text-slate-400";
      bwHeaderSpan.textContent = "B / W";
      headerSwatches.appendChild(bwHeaderSpan);

      headerRow.appendChild(headerSwatches);
      paletteGrid.appendChild(headerRow);

      // Color family rows
      colorFamilies.forEach(family => {
        const row = document.createElement("div");
        row.className = "palette-row";

        const label = document.createElement("div");
        label.className = "palette-family-label";
        label.textContent = family.name;
        row.appendChild(label);

        const swatchesContainer = document.createElement("div");
        swatchesContainer.className = "palette-swatches";

        if (family.name === "black/white") {
          ["black", "white"].forEach(shadeKey => {
            const hex = family.shades[shadeKey];
            const colorName = shadeKey; // "black" or "white"
            const btn = document.createElement("button");
            btn.type = "button";
            btn.className = "palette-swatch";
            btn.style.backgroundColor = hex;
            btn.setAttribute("aria-label", `${family.name} ${shadeKey}`);
            btn.title = `${family.name} ${shadeKey}`;

            const activeColorName = settings[settings.activeColorTarget + "Col"];
            if (activeColorName === colorName) {
              btn.classList.add("selected");
            }

            btn.addEventListener("click", () => {
              applyColorSelection(colorName);
            });
            swatchesContainer.appendChild(btn);
          });
        } else {
          standardShades.forEach(shade => {
            const hex = family.shades[shade];
            const colorName = `${family.name}-${shade}`;
            const btn = document.createElement("button");
            btn.type = "button";
            btn.className = "palette-swatch";
            if (hex) {
              btn.style.backgroundColor = hex;
            }
            btn.setAttribute("aria-label", colorName);
            btn.title = colorName;

            const activeColorName = settings[settings.activeColorTarget + "Col"];
            if (activeColorName === colorName) {
              btn.classList.add("selected");
            }

            btn.addEventListener("click", () => {
              applyColorSelection(colorName);
            });
            swatchesContainer.appendChild(btn);
          });
        }

        row.appendChild(swatchesContainer);
        paletteGrid.appendChild(row);
      });
    }

    function applyColorSelection(colorName) {
      if (settings.activeColorTarget === "bg") {
        settings.bgCol = colorName;
      } else if (settings.activeColorTarget === "start") {
        settings.startCol = colorName;
      } else if (settings.activeColorTarget === "end") {
        settings.endCol = colorName;
      } else if (settings.activeColorTarget === "h1") {
        settings.h1Col = colorName;
      } else if (settings.activeColorTarget === "h2") {
        settings.h2Col = colorName;
      }
      updateHero();
    }


    // ----------------------------------------

    function getCustomColorCss() {
      if (settings.gradient) {
        return `    .hero {
      background-image: linear-gradient(${settings.gradientDirection}, var(--color-${settings.startCol}, ${getColorHex(settings.startCol)}), var(--color-${settings.endCol}, ${getColorHex(settings.endCol)}));
      background-color: transparent;
    }
    .hero h1 {
      color: var(--color-${settings.h1Col}, ${getColorHex(settings.h1Col)});
    }
    .hero h2 {
      color: var(--color-${settings.h2Col}, ${getColorHex(settings.h2Col)});
    }`;
      } else {
        return `    .hero {
      background-image: none;
      background-color: var(--color-${settings.bgCol}, ${getColorHex(settings.bgCol)});
    }
    .hero h1 {
      color: var(--color-${settings.h1Col}, ${getColorHex(settings.h1Col)});
    }
    .hero h2 {
      color: var(--color-${settings.h2Col}, ${getColorHex(settings.h2Col)});
    }`;
      }
    }

    function getColorHex(colorName) {
      if (colorName === "black") return "#000000";
      if (colorName === "white") return "#ffffff";
      const parts = colorName.split("-");
      if (parts.length === 2) {
        const fam = colorFamilies.find(f => f.name === parts[0]);
        if (fam && fam.shades[parts[1]]) {
          return fam.shades[parts[1]];
        }
      }
      return "#000000";
    }

    function updateCustomColorStyles() {
      customColorStyles.textContent = `\n${getCustomColorCss()}\n  `;
    }

      return `    .hero h1 {
      letter-spacing: ${settings.h1LetterSpacing}em;
    }

    .hero h2 {
      letter-spacing: ${settings.h2LetterSpacing}em;
    }`;
    }

    function updateLetterSpacingStyles() {

    function updateHero() {

      // Read current checkbox, select and slider values
      settings.dark = darkCheckbox.checked;
      settings.centre = centreCheckbox.checked;
      settings.minimumHeight = heightCheckbox.checked;
      settings.showSubtitle = subtitleCheckbox.checked;
      settings.gradient = gradientCheckbox.checked;
      settings.gradientDirection = gradientDirectionSelect.value;

      settings.h1Serif = h1SerifCheckbox.checked;
      settings.h2Serif = h2SerifCheckbox.checked;
      settings.h1LetterSpacing = Number(h1SpacingSlider.value);
      settings.h2LetterSpacing = Number(h2SpacingSlider.value);

      // Toggle Gradient Direction and Target visibility based on gradient mode
      if (settings.gradient) {
        gradientDirectionWrapper.style.display = "inline-flex";
        targetBgBtn.style.display = "none";
        targetStartBtn.style.display = "inline-block";
        targetEndBtn.style.display = "inline-block";

        if (settings.activeColorTarget === "bg") {
          settings.activeColorTarget = "start";
        }
      } else {
        gradientDirectionWrapper.style.display = "none";
        targetBgBtn.style.display = "inline-block";
        targetStartBtn.style.display = "none";
        targetEndBtn.style.display = "none";

        if (settings.activeColorTarget === "start" || settings.activeColorTarget === "end") {
          settings.activeColorTarget = "bg";
        }
      }

      // Handle H2 hiding & target selection disabling rules
      if (!settings.showSubtitle) {
        h2SerifCheckbox.disabled = true;
        h2SpacingSlider.disabled = true;
        h2EffectsGroup.classList.add("effect-group-disabled");
        targetH2Btn.disabled = true;

        if (settings.activeColorTarget === "h2") {
          settings.activeColorTarget = settings.gradient ? "start" : "bg";
        }
      } else {
        h2SerifCheckbox.disabled = false;
        h2SpacingSlider.disabled = false;
        h2EffectsGroup.classList.remove("effect-group-disabled");
        targetH2Btn.disabled = false;
      }

      updateTargetButtonsUI();

      // Update hero container classes
      hero.classList.toggle("hero-dark", settings.dark);
      hero.classList.toggle("hero-light", !settings.dark);

      hero.classList.toggle("hero-centre", settings.centre);

      hero.classList.toggle(
        "hero-min-height",
        settings.minimumHeight
      );

      // Update H1 and H2 font classes.
      heroTitle.classList.toggle("heading-serif", settings.h1Serif);
      heroTitle.classList.toggle("heading-sans", !settings.h1Serif);

      heroSubtitle.classList.toggle("heading-serif", settings.h2Serif);
      heroSubtitle.classList.toggle("heading-sans", !settings.h2Serif);

      // Update displayed slider values and dedicated <style> rules.
      h1SpacingValue.textContent = `${settings.h1LetterSpacing}em`;
      h2SpacingValue.textContent = `${settings.h2LetterSpacing}em`;

      updateLetterSpacingStyles();
      updateCustomColorStyles();

      // Hide subheading element
      heroSubtitle.hidden = !settings.showSubtitle;

      // Update active target status text
      let targetNameLabel = "Background";
      if (settings.activeColorTarget === "start") targetNameLabel = "Start colour";
      if (settings.activeColorTarget === "end") targetNameLabel = "End colour";
      if (settings.activeColorTarget === "h1") targetNameLabel = "H1 Colour";
      if (settings.activeColorTarget === "h2") targetNameLabel = "H2 Colour";

      const activeColVal = settings[settings.activeColorTarget + "Col"];
      paletteStatus.textContent = `Active: ${targetNameLabel} (${activeColVal})`;

      // Re-render palette to update selected check/ring indicator
      renderPalette();

      // Keep manual-copy version up to date.
      updateFallbackHtml();
    }

    // ----------------------------------------
    // 5. Keep editing as plain text
    // ----------------------------------------

    function handlePlainTextPaste(event) {
      event.preventDefault();

      const plainText = event.clipboardData
        ? event.clipboardData.getData("text/plain")
        : "";

      const selection = window.getSelection();

      if (!selection || selection.rangeCount === 0) {
        return;
      }

      const range = selection.getRangeAt(0);
      const heading = event.currentTarget;

      if (!heading.contains(range.commonAncestorContainer) &&
          range.commonAncestorContainer !== heading) {
        return;
      }

      range.deleteContents();

      const textNode = document.createTextNode(plainText);
      range.insertNode(textNode);

      range.setStartAfter(textNode);
      range.collapse(true);

      selection.removeAllRanges();
      selection.addRange(range);

      heading.dispatchEvent(new Event("input", {
        bubbles: true
      }));
    }


    function preventNewLines(event) {
      if (event.key === "Enter") {
        event.preventDefault();
      }
    }


    function handleHeadingInput() {
      updateFallbackHtml();
    }


    // ----------------------------------------
    // 6. Escape text for safe HTML output
    // ----------------------------------------

    function escapeHtml(text) {
      return text
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }


    // ----------------------------------------
    // 7. Generate the standalone hero page
    // ----------------------------------------

    function generateHtml() {
      const titleText = escapeHtml(heroTitle.textContent);
      const subtitleText = escapeHtml(heroSubtitle.textContent);

      const heroClasses = [
        "hero",
        settings.dark ? "hero-dark" : "hero-light",
        settings.centre ? "hero-centre" : "",
        settings.minimumHeight ? "hero-min-height" : ""
      ].filter(Boolean).join(" ");

      const titleFontClass = settings.h1Serif
        ? "heading-serif"
        : "heading-sans";

      const subtitleFontClass = settings.h2Serif
        ? "heading-serif"
        : "heading-sans";

      const subtitleHtml = settings.showSubtitle
        ? `\n      <h2 class="${subtitleFontClass}">${subtitleText}</h2>`
        : "";

      const heroStyles = document.getElementById("hero-styles").textContent;
      const customColorCss = getCustomColorCss();
      const letterSpacingCss = getLetterSpacingCss();

      return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>My Hero Section</title>
  <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">

  <script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>

  <style type="text/tailwindcss">
${heroStyles}
  </style>

  <style>
${customColorCss}

${letterSpacingCss}
  </style>
</head>
<body>

  <header class="${heroClasses}">
    <div class="hero-content">
      <h1 class="${titleFontClass}">${titleText}</h1>${subtitleHtml}
    </div>
  </header>

</body>
</html>`;
    }


    // ----------------------------------------
    // 8. Update the manual-copy textarea
    // ----------------------------------------

    function updateFallbackHtml() {
      generatedHtml.value = generateHtml();
    }


    // ----------------------------------------
    // 9. Copy HTML using the Clipboard API
    // ----------------------------------------

    async function copyHtml() {
      const html = generateHtml();
      copyStatus.textContent = "";

      try {
        if (!navigator.clipboard ||
            typeof navigator.clipboard.writeText !== "function") {
          throw new Error("Clipboard API unavailable");
        }

        await navigator.clipboard.writeText(html);

        copyStatus.textContent =
          "HTML copied—paste it into VS Code.";

        fallbackSection.hidden = true;

      } catch (error) {
        generatedHtml.value = html;
        fallbackSection.hidden = false;

        copyStatus.textContent =
          "Automatic copying is unavailable. Select the HTML below and copy it manually.";
      }
    }
